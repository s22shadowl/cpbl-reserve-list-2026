## P0-1 可行性驗證結果（2026-11-15 已完成）

- **不需要 Playwright**。官網在 HiNet CDN 後面：第一次請求會回 `308`，並發一個 `__chtcdn` cookie，再轉址回同一個網址。用 `requests.Session` 保存 cookie 就能正常取得頁面。原作者說「官網不讓爬」，應該就是卡在這裡。
- **一軍名單**：`GET /team?ClubNo=<code>`。
- **二軍名單**：`POST /team`，表單欄位是 `__RequestVerificationToken`（取自一軍頁的 hidden input）、`ClubNo` 和 `KindCode=D`。
- 名單項目的 selector 是 `.TeamPlayersList .item`。
  - 球員有 `.cont .name a[href*=Acnt=]`；教練沒有連結，所以會被自然略過。
  - 其他欄位：`.cont .pos` 是守位文字，`.cont .number` 是背號，照片在 `.img a` 的 `style` 裡的 `background-image:url(...)`，原圖是 300×385。
- **名字前綴**：官網頁面上的圖例如下。
  - `＃` 註銷註冊：仍是合約球員，**可選**。
  - `✽` 合約所屬：**可選**。
  - `◎` 自主培訓、自行培訓、測試洋將：**排除**。
- **個人頁**：`GET /team/person?Acnt=<id>`，資料在 `.PlayerBrief dd.<class> .desc`。
  - `born` 的格式是 `1997/12/29`。
  - `nationality` 是「中華民國」或其他國家，用來自動判斷洋將。
  - `draft` 的格式是「富邦悍將2019年第一輪」，洋將這一欄是空的。
- **六隊實測（一軍加二軍）**：ACN 88、ADD 79、AJL 81、AEO 96、AAA 81、AKP 76 人。守位文字都在對照表內，也都有照片網址。
- 原版的悍將資料有 86 人可選，排除了 10 人（6 名洋將、3 名 ◎、1 名 `#` 但已非球員），和官網的 96 人吻合。
- 測試 fixture 已經放在 `scraper/tests/fixtures/`：
  - `team_AEO_A.html`、`team_AEO_D.html`
  - `person_0000005479.html`（江國豪，本土球員）
  - `person_0000006749.html`（魔力藍，美國籍）

```python
"""Parse CPBL team player-list pages (GET /team?ClubNo=X and POST /team KindCode=D)."""

import re
from typing import List, Optional

from bs4 import BeautifulSoup, Tag

from .models import RosterEntry

# Official legend on the team page:
#   ＃ 註銷註冊 (still a contract player)   ✽ 合約所屬球員 (farm only)
#   ◎ 自主培訓、自行培訓、測試洋將、洋將（不占洋將名額）
NAME_MARKS = {"◎": "◎", "#": "#", "＃": "#", "*": "*", "✽": "*", "＊": "*"}
_ACNT = re.compile(r"Acnt=(\d{10})")
_BG_URL = re.compile(r"url\(['\"]?([^'\")]+)['\"]?\)")


def parse_token(html: str) -> str:
    """Return the anti-forgery token needed to POST /team."""
    soup = BeautifulSoup(html, "lxml")
    field = soup.find("input", attrs={"name": "__RequestVerificationToken"})
    if not isinstance(field, Tag) or not field.get("value"):
        raise ValueError("找不到 __RequestVerificationToken")
    return str(field["value"])


def _split_mark(raw: str) -> tuple:
    raw = raw.strip()
    if raw and raw[0] in NAME_MARKS:
        return NAME_MARKS[raw[0]], raw[1:].strip()
    return "", raw


def _photo_url(item: Tag) -> Optional[str]:
    img = item.select_one(".img a, .img span")
    style = img.get("style", "") if isinstance(img, Tag) else ""
    m = _BG_URL.search(style)
    return m.group(1) if m else None


def _text(item: Tag, selector: str) -> str:
    node = item.select_one(selector)
    return node.get_text(strip=True) if node else ""


def parse_roster(html: str, level: str) -> List[RosterEntry]:
    """Return players (coaches skipped: they have no person link) in page order."""
    soup = BeautifulSoup(html, "lxml")
    entries = []
    for item in soup.select(".TeamPlayersList .item"):
        link = item.select_one(".cont .name a[href]")
        if not isinstance(link, Tag):
            continue
        m = _ACNT.search(link["href"])
        if not m:
            continue
        mark, name = _split_mark(link.get_text())
        entries.append(
            RosterEntry(
                id=m.group(1),
                name=name,
                mark=mark,
                no=_text(item, ".cont .number"),
                pos=_text(item, ".cont .pos"),
                level=level,
                photo_url=_photo_url(item),
            )
        )
    return entries
```

```python
"""Parse a CPBL player personal page (GET /team/person?Acnt=X)."""

import re
from typing import Optional

from bs4 import BeautifulSoup

from .models import PersonInfo

_CN_DIGITS = {"一": 1, "二": 2, "三": 3, "四": 4, "五": 5, "六": 6, "七": 7, "八": 8, "九": 9}
_DRAFT = re.compile(r"(\d{4})年第([0-9一二三四五六七八九十]+)輪")


def cn_number(text: str) -> int:
    """Convert 1-99 written as digits or Chinese numerals ("十二", "二十") to int."""
    if text.isdigit():
        return int(text)
    if "十" not in text:
        return _CN_DIGITS[text]
    tens, _, ones = text.partition("十")
    return (_CN_DIGITS[tens] if tens else 1) * 10 + (_CN_DIGITS[ones] if ones else 0)


def _desc(soup: BeautifulSoup, cls: str) -> str:
    node = soup.select_one(f".PlayerBrief dd.{cls} .desc")
    return node.get_text(strip=True) if node else ""


def parse_draft(text: str) -> tuple:
    """'富邦悍將2019年第一輪' -> (2019, 1); anything else -> (None, None)."""
    m = _DRAFT.search(text)
    if not m:
        return None, None
    return int(m.group(1)), cn_number(m.group(2))


def parse_person(html: str) -> PersonInfo:
    soup = BeautifulSoup(html, "lxml")
    if soup.select_one(".PlayerBrief") is None:
        raise ValueError("找不到 PlayerBrief 區塊")
    born_raw = _desc(soup, "born")
    born: Optional[str] = born_raw.replace("/", "-") if re.fullmatch(r"\d{4}/\d{2}/\d{2}", born_raw) else None
    year, rnd = parse_draft(_desc(soup, "draft"))
    return PersonInfo(born=born, nationality=_desc(soup, "nationality"), draft_year=year, draft_round=rnd)
```

```python
"""Decide each player's position group and whether they are exempt from the 60-man list."""

import json
import re
from pathlib import Path
from typing import Optional, Tuple

from .models import PersonInfo, RosterEntry

REASONS = ("foreign", "training", "fa", "expansion", "removed")
DOMESTIC = "中華民國"
_INFIELD = {"一壘手", "二壘手", "三壘手", "游擊手", "內野手"}
_ID = re.compile(r"^\d{10}$")


def position_group(pos: str) -> Tuple[str, Optional[str]]:
    """'投手' -> ('P', None); '捕手' -> ('C', '捕手'); '二壘手' -> ('I', '二壘手'); '左外野手' -> ('O', '左外野手')."""
    if pos == "投手":
        return "P", None
    if pos == "捕手":
        return "C", pos
    if pos in _INFIELD:
        return "I", pos
    if "外野" in pos:
        return "O", pos
    raise ValueError(f"未知守位：{pos}")


def load_overrides(path: Path) -> dict:
    """Read and validate overrides.json: {"include": [id...], "exclude": {id: {"reason", "name", "note"}}}."""
    data = json.loads(path.read_text(encoding="utf-8"))
    include = data.get("include", [])
    exclude = data.get("exclude", {})
    for pid in [*include, *exclude]:
        if not _ID.match(pid):
            raise ValueError(f"overrides 球員 ID 格式錯誤：{pid}")
    for pid, rule in exclude.items():
        if rule.get("reason") not in REASONS:
            raise ValueError(f"overrides {pid} 的 reason 必須是 {REASONS} 之一")
    overlap = set(include) & set(exclude)
    if overlap:
        raise ValueError(f"overrides 同時 include 與 exclude：{sorted(overlap)}")
    return {"include": frozenset(include), "exclude": exclude}


def exempt_reason(entry: RosterEntry, person: PersonInfo, overrides: dict) -> Optional[str]:
    """None when the player must be chosen from; otherwise one of REASONS.

    Priority: manual include > manual exclude > non-domestic nationality > ◎ mark.
    """
    if entry.id in overrides["include"]:
        return None
    if entry.id in overrides["exclude"]:
        return overrides["exclude"][entry.id]["reason"]
    if person.nationality and person.nationality != DOMESTIC:
        return "foreign"
    if entry.mark == "◎":
        return "training"
    return None
```

```python
"""Turn parsed records into the docs/data/teams.json and worker/src/roster.json shapes."""

from typing import Dict, Iterable, Mapping, Tuple

from .classify import REASONS, exempt_reason, position_group
from .models import PersonInfo, RosterEntry


def player_record(entry: RosterEntry, person: PersonInfo, has_photo: bool) -> dict:
    g, p = position_group(entry.pos)
    rec = {"id": entry.id, "n": entry.name, "no": entry.no, "g": g, "l": entry.level}
    if p:
        rec["p"] = p
    rec["ph"] = 1 if has_photo else 0
    optional = {"b": person.born, "dy": person.draft_year, "dr": person.draft_round}
    rec.update({k: v for k, v in optional.items() if v is not None})
    return rec


def build_team(
    entries: Iterable[RosterEntry],
    persons: Mapping[str, PersonInfo],
    overrides: dict,
    photo_ids: frozenset,
) -> dict:
    players = []
    exempt: Dict[str, list] = {r: [] for r in REASONS}
    for e in entries:
        reason = exempt_reason(e, persons[e.id], overrides)
        if reason:
            exempt[reason].append(e.name)
        else:
            players.append(player_record(e, persons[e.id], e.id in photo_ids))
    return {"players": players, "exempt": exempt}


def build_outputs(teams: Mapping[str, dict], updated: str) -> Tuple[dict, dict]:
    teams_json = {"updated": updated, "teams": dict(teams)}
    roster_json = {code: sorted(p["id"] for p in t["players"]) for code, t in teams.items()}
    return teams_json, roster_json
```

```python
from io import BytesIO

from PIL import Image

PHOTO_WIDTH = 240
JPEG_QUALITY = 82


def resize_photo(data: bytes) -> bytes:
    """Return a JPEG no wider than PHOTO_WIDTH, keeping the aspect ratio."""
    with Image.open(BytesIO(data)) as img:
        rgb = img.convert("RGB")
    if rgb.width > PHOTO_WIDTH:
        height = round(rgb.height * PHOTO_WIDTH / rgb.width)
        rgb = rgb.resize((PHOTO_WIDTH, height), Image.LANCZOS)
    out = BytesIO()
    rgb.save(out, "JPEG", quality=JPEG_QUALITY, optimize=True)
    return out.getvalue()
```

```python
"""HTTP access to www.cpbl.com.tw, ported from cpbl_takao_today_BE app/core/fetcher.py.

The site sits behind HiNet CDN: the first request answers 308 with a __chtcdn cookie and
redirects to the same URL. A requests.Session keeps that cookie, so plain requests works.
"""

import logging
import time
from typing import Callable, Optional
from urllib.parse import quote

import requests

from .parse_team import parse_token

BASE_URL = "https://www.cpbl.com.tw"
USER_AGENT = (
    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
    "(KHTML, like Gecko) Chrome/130.0 Safari/537.36"
)
TIMEOUT = 30
DELAY_SECONDS = 1.5
MAX_ATTEMPTS = 3

log = logging.getLogger(__name__)


class RetryableScraperError(Exception):
    """Temporary failure (5xx, timeout, connection); worth retrying."""


class FatalScraperError(Exception):
    """Permanent failure (4xx, unexpected page); do not retry."""


class CpblClient:
    def __init__(
        self,
        session: Optional[requests.Session] = None,
        delay: float = DELAY_SECONDS,
        sleep: Callable[[float], None] = time.sleep,
    ):
        self._session = session or requests.Session()
        self._session.headers.update({"User-Agent": USER_AGENT})
        self._delay = delay
        self._sleep = sleep

    def _once(self, method: str, url: str, **kwargs) -> requests.Response:
        try:
            resp = self._session.request(method, url, timeout=TIMEOUT, **kwargs)
        except requests.RequestException as e:
            raise RetryableScraperError(f"{method} {url} 失敗：{e}") from e
        if 500 <= resp.status_code < 600:
            raise RetryableScraperError(f"{method} {url} 伺服器錯誤 HTTP {resp.status_code}")
        if resp.status_code >= 400:
            raise FatalScraperError(f"{method} {url} 客戶端錯誤 HTTP {resp.status_code}")
        return resp

    def _request(self, method: str, path: str, **kwargs) -> requests.Response:
        url = BASE_URL + path
        for attempt in range(1, MAX_ATTEMPTS + 1):
            self._sleep(self._delay)  # friendly pacing before every request
            try:
                return self._once(method, url, **kwargs)
            except RetryableScraperError as e:
                if attempt == MAX_ATTEMPTS:
                    raise
                log.warning("第 %d 次失敗，稍後重試：%s", attempt, e)
                self._sleep(self._delay * 2 ** attempt)
        raise AssertionError("unreachable")

    def team_pages(self, code: str) -> tuple:
        """Return (一軍 HTML, 二軍 HTML) for a team code such as 'AEO'."""
        first = self._request("GET", f"/team?ClubNo={code}").text
        farm = self._request(
            "POST",
            "/team",
            data={"__RequestVerificationToken": parse_token(first), "ClubNo": code, "KindCode": "D"},
            headers={"X-Requested-With": "XMLHttpRequest", "Referer": f"{BASE_URL}/team?ClubNo={code}"},
        ).text
        return first, farm

    def person(self, acnt: str) -> str:
        return self._request("GET", f"/team/person?Acnt={acnt}").text

    def photo(self, path: str) -> bytes:
        return self._request("GET", quote(path)).content
```

```python
"""Validate generated data before committing: python -m scraper.check"""

import json
import logging
import re
import sys
from pathlib import Path
from typing import List, Tuple

from .classify import REASONS
from .models import TEAM_CODES


LIMIT = 60
REPO = Path(__file__).resolve().parent.parent
_ID = re.compile(r"^\d{10}$")

log = logging.getLogger(__name__)


def check(teams_json: dict, roster_json: dict, photo_ids: frozenset) -> Tuple[List[str], List[str]]:
    errors: List[str] = []
    warnings: List[str] = []
    teams = teams_json.get("teams", {})
    if not re.match(r"^\d{4}-\d{2}-\d{2}$", str(teams_json.get("updated", ""))):
        errors.append("updated 必須是 YYYY-MM-DD")
    for code in TEAM_CODES:
        if code not in teams:
            errors.append(f"{code}：teams.json 缺少此隊")
            continue
        players = teams[code].get("players", [])
        ids = [p["id"] for p in players]
        bad = [i for i in ids if not _ID.match(i)]
        if bad:
            errors.append(f"{code}：ID 格式錯誤 {bad}")
        if len(set(ids)) != len(ids):
            errors.append(f"{code}：球員 ID 重複")
        if len(ids) <= LIMIT:
            errors.append(f"{code}：可選球員只有 {len(ids)} 人，應多於 {LIMIT}")
        if sorted(ids) != sorted(roster_json.get(code, [])):
            errors.append(f"{code}：teams.json 與 roster.json 的球員不一致")
        if set(teams[code].get("exempt", {})) != set(REASONS):
            errors.append(f"{code}：exempt 欄位必須剛好是 {REASONS}")
        for p in players:
            if p.get("ph") and p["id"] not in photo_ids:
                errors.append(f"{code}：{p['n']} 標記有照片但檔案不存在")
            if not p.get("ph"):
                warnings.append(f"{code}：{p['n']} 沒有照片")
    extra = set(roster_json) - set(TEAM_CODES)
    if extra:
        errors.append(f"roster.json 有未知球隊 {sorted(extra)}")
    return errors, warnings


def main() -> int:
    logging.basicConfig(level=logging.INFO, format="%(message)s")
    teams_json = json.loads((REPO / "docs/data/teams.json").read_text(encoding="utf-8"))
    roster_json = json.loads((REPO / "worker/src/roster.json").read_text(encoding="utf-8"))
    photo_ids = frozenset(p.stem for p in (REPO / "docs/photos").glob("*.jpg"))
    errors, warnings = check(teams_json, roster_json, photo_ids)
    for w in warnings:
        log.warning("警告：%s", w)
    for e in errors:
        log.error("錯誤：%s", e)
    counts = {c: len(t["players"]) for c, t in teams_json.get("teams", {}).items()}
    log.info("可選人數：%s；%d 個錯誤、%d 個警告", counts, len(errors), len(warnings))
    return 1 if errors else 0


if __name__ == "__main__":
```

```python
"""Scrape cpbl.com.tw and regenerate site data: python -m scraper.build [--team AEO ...] [--refresh]"""

import argparse
import json
import logging
import sys
from datetime import date
from pathlib import Path
from typing import Dict, Iterable

from .classify import exempt_reason, load_overrides
from .fetcher import CpblClient, FatalScraperError, RetryableScraperError
from .models import TEAM_CODES, PersonInfo, RosterEntry
from .output import build_outputs, build_team
from .parse_person import parse_person
from .parse_team import parse_roster
from .photos import resize_photo


REPO = Path(__file__).resolve().parent.parent
TEAMS_JSON = REPO / "docs/data/teams.json"
ROSTER_JSON = REPO / "worker/src/roster.json"
PHOTOS = REPO / "docs/photos"
CACHE = REPO / "scraper/.cache/person"
OVERRIDES = REPO / "scraper/overrides.json"

log = logging.getLogger(__name__)


def person_html(client: CpblClient, acnt: str, cache: Path, refresh: bool) -> str:
    """Fetch a personal page, reusing scraper/.cache unless refresh is set."""
    path = cache / f"{acnt}.html"
    if path.exists() and not refresh:
        return path.read_text(encoding="utf-8")
    html = client.person(acnt)
    cache.mkdir(parents=True, exist_ok=True)
    path.write_text(html, encoding="utf-8")
    return html


def ensure_photo(client: CpblClient, entry: RosterEntry, photos: Path, refresh: bool) -> bool:
    """Download and shrink a player's photo; False (and a warning) when unavailable."""
    path = photos / f"{entry.id}.jpg"
    if path.exists() and not refresh:
        return True
    if not entry.photo_url:
        log.warning("%s %s 官網沒有照片", entry.id, entry.name)
        return False
    try:
        data = resize_photo(client.photo(entry.photo_url))
    except (RetryableScraperError, FatalScraperError, OSError) as e:
        log.warning("%s %s 照片下載失敗：%s", entry.id, entry.name, e)
        return False
    photos.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)
    return True


def scrape_team(client: CpblClient, code: str, overrides: dict, photos: Path, cache: Path, refresh: bool) -> dict:
    first_html, farm_html = client.team_pages(code)
    entries = parse_roster(first_html, "一軍") + parse_roster(farm_html, "二軍")
    if not entries:
        raise FatalScraperError(f"{code}：球員列表是空的，頁面結構可能改變")
    persons: Dict[str, PersonInfo] = {}
    photo_ids = set()
    for e in entries:
        persons[e.id] = parse_person(person_html(client, e.id, cache, refresh))
        if exempt_reason(e, persons[e.id], overrides) is None and ensure_photo(client, e, photos, refresh):
            photo_ids.add(e.id)
    team = build_team(entries, persons, overrides, frozenset(photo_ids))
    log.info("%s：一二軍共 %d 人，可選 %d 人", code, len(entries), len(team["players"]))
    return team


def run(client: CpblClient, codes: Iterable[str], refresh: bool, today: str,
        teams_path: Path = TEAMS_JSON, roster_path: Path = ROSTER_JSON,
        photos: Path = PHOTOS, cache: Path = CACHE, overrides_path: Path = OVERRIDES) -> None:
    overrides = load_overrides(overrides_path)
    existing = json.loads(teams_path.read_text(encoding="utf-8"))["teams"] if teams_path.exists() else {}
    fresh = {code: scrape_team(client, code, overrides, photos, cache, refresh) for code in codes}
    merged = {code: fresh.get(code, existing.get(code)) for code in TEAM_CODES if code in fresh or code in existing}
    teams_json, roster_json = build_outputs(merged, today)
    teams_path.parent.mkdir(parents=True, exist_ok=True)
    teams_path.write_text(json.dumps(teams_json, ensure_ascii=False, separators=(",", ":")), encoding="utf-8")
    roster_path.write_text(json.dumps(roster_json, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="從中職官網產生 teams.json、roster.json 與照片")
    parser.add_argument("--team", action="append", choices=TEAM_CODES, help="只更新指定球隊，可重複；預設六隊")
    parser.add_argument("--refresh", action="store_true", help="忽略個人頁快取並重新下載照片")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
    try:
        run(CpblClient(), args.team or TEAM_CODES, args.refresh, date.today().isoformat())
    except (FatalScraperError, RetryableScraperError, ValueError) as e:
        log.error("產生失敗：%s", e)
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
